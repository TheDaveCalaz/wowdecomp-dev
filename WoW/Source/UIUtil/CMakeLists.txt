file(GLOB_RECURSE PRIVATE_SOURCES CONFIGURE_DEPENDS "*.cpp")

add_library(UIUtil STATIC
    ${PRIVATE_SOURCES}
)

target_include_directories(UIUtil
    PUBLIC
        ${CMAKE_SOURCE_DIR}
)

# Not in the original client: the modern addon layer (ModernUI/*.lua) is built into the exe.
# Every .lua file becomes a byte array in ModernUI_Embedded.h, run in file-name order.
file(GLOB MODERNUI_LUA CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/ModernUI/*.lua")
list(SORT MODERNUI_LUA)
set(MODERNUI_HEADER "${CMAKE_CURRENT_BINARY_DIR}/generated/ModernUI_Embedded.h")
set(MODERNUI_BODY "// Generated by CMake from WoW/Source/UIUtil/ModernUI. Do not edit.\n#pragma once\n\n")
set(MODERNUI_TABLE "")
set(MODERNUI_INDEX 0)
foreach (lua_file ${MODERNUI_LUA})
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${lua_file}")
    get_filename_component(lua_name "${lua_file}" NAME)
    file(READ "${lua_file}" lua_hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f])" "\\1\n" lua_hex "${lua_hex}")
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," lua_hex "${lua_hex}")
    string(APPEND MODERNUI_BODY "static const unsigned char s_modernUI_${MODERNUI_INDEX}[] = { ${lua_hex}0x00 };\n")
    string(APPEND MODERNUI_TABLE "    { \"Interface\\\\ModernUI\\\\${lua_name}\", reinterpret_cast<const char*>(s_modernUI_${MODERNUI_INDEX}) },\n")
    math(EXPR MODERNUI_INDEX "${MODERNUI_INDEX} + 1")
endforeach ()
string(APPEND MODERNUI_BODY "\nstruct ModernUIFile { const char* name; const char* source; };\n")
string(APPEND MODERNUI_BODY "static const ModernUIFile s_modernUIFiles[] = {\n${MODERNUI_TABLE}    { nullptr, nullptr },\n};\n")
file(WRITE "${MODERNUI_HEADER}.tmp" "${MODERNUI_BODY}")
configure_file("${MODERNUI_HEADER}.tmp" "${MODERNUI_HEADER}" COPYONLY)

target_include_directories(UIUtil
    PRIVATE
        ${CMAKE_CURRENT_BINARY_DIR}/generated
)
